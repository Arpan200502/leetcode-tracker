1class Solution {
2    public boolean checkValidString(String s) {
3        Stack<Integer> open=new Stack<>();
4        Stack<Integer> star=new Stack<>();
5
6        for(int i=0;i<s.length();i++){
7            char ch=s.charAt(i);
8
9            if(ch=='('){
10                open.push(i);
11            }else if(ch=='*'){
12                star.push(i);
13            }else{
14                if(!open.isEmpty()){
15                    open.pop();
16                }else if(!star.isEmpty()){
17                    star.pop();
18                }else{
19                    return false;
20                }
21            }
22        }
23
24        while(!open.isEmpty() && !star.isEmpty()){
25            if(open.peek()<star.peek()){
26                open.pop();
27                star.pop();
28            }else{
29                return false;
30            }
31        }
32
33        return open.isEmpty();
34
35    }
36}
